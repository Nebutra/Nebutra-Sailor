## 安装

```bash
pip install nebutra-sdk
```

需要 Python 3.11+。

## 初始化客户端

```python
import os
from nebutra import Client

nebutra = Client(
    api_key=os.environ["NEBUTRA_API_KEY"],
    org_id=os.environ["NEBUTRA_ORG_ID"],
)
```

## 错误处理

```python
from nebutra import NebutraError, NotFoundError, ValidationError, RateLimitError
import asyncio

try:
    project = await nebutra.projects.get("proj_123")
except NotFoundError:
    pass  # 404 — 资源未找到
except ForbiddenError:
    pass  # 403 — 权限不足
except ValidationError as e:
    for field in e.fields:
        print(f"{field.field}: {field.message}")
except RateLimitError as e:
    await asyncio.sleep(e.retry_after)
except NebutraError as e:
    print(f"[{e.code}] {e.message} (请求 ID: {e.request_id})")
```

## FastAPI 集成

```python
from fastapi import FastAPI, Depends, HTTPException, Header
from nebutra import Client, verify_token
import os

nebutra = Client(api_key=os.environ["NEBUTRA_API_KEY"], org_id=os.environ["NEBUTRA_ORG_ID"])

async def get_current_auth(authorization: str = Header(...)):
    token = authorization.removeprefix("Bearer ")
    try:
        return await verify_token(token, api_key=os.environ["NEBUTRA_API_KEY"])
    except Exception:
        raise HTTPException(status_code=401, detail="无效令牌")

app = FastAPI()

@app.get("/api/projects")
async def list_projects(auth=Depends(get_current_auth)):
    projects = await nebutra.projects.list(org_id=auth.org_id)
    return {"data": projects.data}
```
